3. Core Technical Specifications


A. Custom Lightweight Markup & 20-Language Code Parser

Your custom parser runs before Markdown/MDX rendering to normalize student notes and handwritten transcriptions into structured Abstract Syntax Tree (AST) blocks:
• Title (***): Any line followed immediately by ^\*{3,}$ becomes an <h1> Document Title.
• Section (===): Any line followed immediately by ^={3,}$ becomes an <h2> Section Heading.
• Subsection (---): Any line followed immediately by ^-{3,}$ becomes an <h3> Subsection Heading.
• Indented Code Blocks (   ): Consecutive lines prefixed by 3 spaces (^ {3}(.*)$) are grouped into a fenced code block. An automatic language classifier (using token heuristics or Tree-sitter) tags the block among the 20 supported languages (Python, JavaScript, TypeScript, Java, C#, Kotlin, Dart/Flutter, Bash/Shell, Rust, Go, C++, C, PHP, Ruby, PowerShell, SQL, R, Scala, MATLAB, COBOL/Assembly).

• Supported Languages (i18n & OCR): English (en), French (fr), German (de), Italian (it), Spanish (es), Portuguese (pt), Indonesian (id), Swahili (sw), Amharic (am), Hausa (ha), Turkish (tr), Russian (ru), Chinese (zh), Japanese (ja), Korean (ko), Arabic (ar), Urdu (ur), Farsi/Persian (fa), Western Punjabi (pa-PK), Tagalog/Filipino (tl), Vietnamese (vi), Hindi (hi), Marathi (mr), Bangla/Bengali (bn), Tamil (ta), and Telugu (te).

Example: Custom Lightweight Markup Parser (setextCustomParser.ts)

typescript
export function parseCustomLightweightMarkup(raw: string): string {
  const lines = raw.replace(/\r\n/g, '\n').split('\n');
  const output: string[] = [];
  let i = 0;

  while (i < lines.length) {
    const current = lines[i];
    const next = lines[i + 1] ?? '';

    if (current.trim().length > 0 && /^\*{3,}$/.test(next.trim())) {
      output.push(`# ${current.trim()}`);
      i += 2;
      continue;
    }
    if (current.trim().length > 0 && /^={3,}$/.test(next.trim())) {
      output.push(`## ${current.trim()}`);
      i += 2;
      continue;
    }
    if (current.trim().length > 0 && /^-{3,}$/.test(next.trim())) {
      output.push(`### ${current.trim()}`);
      i += 2;
      continue;
    }
    if (/^ {3}\S/.test(current) || (current.startsWith('   ') && current.trim() !== '')) {
      const codeLines: string[] = [];
      while (i < lines.length && (lines[i].startsWith('   ') || lines[i].trim() === '')) {
        codeLines.push(lines[i].startsWith('   ') ? lines[i].slice(3) : '');
        i++;
      }
      output.push('```auto\n' + codeLines.join('\n').trimEnd() + '\n```');
      continue;
    }

    output.push(current);
    i++;
  }

  return output.join('\n');
}


Example: Rebindable Keyboard Shortcuts Component (KeyboardShortcuts.tsx)

tsx
import React, { useState, useEffect } from 'react';

export interface ShortcutItem {
  id: string;
  action: string;
  keys: string;
}

const DEFAULT_SHORTCUTS: ShortcutItem[] = [
  { id: 'global-search', action: 'Open Global Search', keys: 'Ctrl+K' },
  { id: 'scan-capture', action: 'Scan / Upload Handwriting', keys: 'Ctrl+Shift+U' },
  { id: 'export-pdf', action: 'Export Multilingual PDF', keys: 'Ctrl+E' },
  { id: 'ai-quiz', action: 'Generate Study Quiz', keys: 'Ctrl+Alt+Q' },
  { id: 'user-guide', action: 'Open User Guide Modal', keys: 'F1' },
];

export const KeyboardShortcuts: React.FC = () => {
  const [shortcuts, setShortcuts] = useState<ShortcutItem[]>(() => {
    const saved = localStorage.getItem('ainotes_shortcuts');
    return saved ? JSON.parse(saved) : DEFAULT_SHORTCUTS;
  });
  const [recordingId, setRecordingId] = useState<string | null>(null);

  useEffect(() => {
    localStorage.setItem('ainotes_shortcuts', JSON.stringify(shortcuts));
  }, [shortcuts]);

  const handleKeyDown = (e: React.KeyboardEvent, id: string) => {
    e.preventDefault();
    const parts: string[] = [];
    if (e.ctrlKey) parts.push('Ctrl');
    if (e.metaKey) parts.push('Cmd');
    if (e.altKey) parts.push('Alt');
    if (e.shiftKey) parts.push('Shift');
    if (!['Control', 'Meta', 'Alt', 'Shift'].includes(e.key)) {
      parts.push(e.key.length === 1 ? e.key.toUpperCase() : e.key);
      const combo = parts.join('+');
      setShortcuts((prev) =>
        prev.map((item) => (item.id === id ? { ...item, keys: combo } : item))
      );
      setRecordingId(null);
    }
  };

  return (
    <div className="p-4 border rounded-lg max-w-xl">
      <div className="flex justify-between items-center mb-3">
        <h3 className="text-lg font-semibold">Keyboard Shortcuts</h3>
        <button
          onClick={() => setShortcuts(DEFAULT_SHORTCUTS)}
          className="text-sm underline"
        >
          Reset Defaults
        </button>
      </div>
      <ul className="divide-y">
        {shortcuts.map((sc) => (
          <li key={sc.id} className="py-2 flex justify-between items-center">
            <span>{sc.action}</span>
            {recordingId === sc.id ? (
              <input
                autoFocus
                readOnly
                placeholder="Press keys..."
                onKeyDown={(e) => handleKeyDown(e, sc.id)}
                onBlur={() => setRecordingId(null)}
                className="px-2 py-1 border rounded text-sm bg-yellow-50"
              />
            ) : (
              <button
                onClick={() => setRecordingId(sc.id)}
                className="px-2 py-1 border rounded text-sm font-mono hover:bg-gray-100"
              >
                {sc.keys}
              </button>
            )}
          </li>
        ))}
      </ul>
    </div>
  );
};
